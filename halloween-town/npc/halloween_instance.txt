// Portrait pop-up: 1 = Casper (bottom-right, like Nightwing),
// 2 = Blair (bottom-left, like Leticia), 3 = Jakk the Halloween Exchanger
// (bottom-left), 0 = hide.
// Files: data/texture/ui/illust/hw_casper.bmp, hw_blair.bmp, hw_jakk.bmp
function	script	HW_Portrait	{
	switch (getarg(0)) {
	case 1:  cutin "hw_casper.bmp", 2; break;
	case 2:  cutin "hw_blair.bmp", 0; break;
	case 3:  cutin "hw_jakk.bmp", 0; break;
	default: cutin "", 255;
	}
	return;
}

// A navigation link to a spot on hw_town: callfunc("HW_Navi", "<text>", x, y).
// Clicking it opens the navigation; the coordinates are never shown.
function	script	HW_Navi	{
	return "<NAVI>" + getarg(0) + "<INFO>hw_town," + getarg(1) + "," + getarg(2) + ",0,000,0</INFO></NAVI>";
}

// 1 when the attached character belongs to the open run: its owner, or a
// member of the owner's party who has joined it (talked to Casper)
function	script	HW_InRun	{
	if (getcharid(0) == $@HW_CID) return 1;
	return ($@HW_PID && getcharid(1) == $@HW_PID && HW_Run == $@HW_RunID);
}

// mm:ss for <seconds>
function	script	HW_Clock	{
	.@t = getarg(0);
	return (.@t / 60) + ":" + ((.@t % 60) < 10 ? "0" : "") + (.@t % 60);
}

// A Lost Child is found: any member of the run can send them home.
//   callfunc "HW_Child", <1..6>;   (from the child's own NPC)
function	script	HW_Child	{
	.@c = getarg(0);
	if (!$@HW_Active || !callfunc("HW_InRun") || !($@HW_KidOut & (1 << .@c))) return;
	setarray .@line$[1],
		"Is it morning yet? Mommy said to wait by the pumpkins until the bell rang... the bell rang a long time ago.",
		"I was a vampire for Halloween! A real scary one! ...Then the shadows came, and now I'm a real ghost. I don't like it.",
		"I hid all my candy so the big kids wouldn't take it. Now I can't find my candy OR my house.",
		"Are you a hero? Casper says heroes always come back for the little ones. I've been counting... I counted to a million.",
		"My big sister ran to Umbala. She said she'd come back for me. Is she coming?",
		"I'm not scared. I'm NOT. ...Can you hold my hand anyway? Ghosts can hold hands, right?";
	mes "[Lost Child]";
	mes .@line$[.@c];
	next;
	callfunc "HW_Portrait", 1;
	mes "[Casper]";
	mes "Shh, it's okay. Follow the moonlight, little one: it goes all the way home. Your family is waiting on the other side.";
	callfunc "HW_Portrait", 0;
	close2;
	if (!($@HW_KidOut & (1 << .@c))) return;	// someone else got there first
	$@HW_KidOut = $@HW_KidOut & ~(1 << .@c);
	$@HW_Kids++;
	$@HW_Bless = 1;	// the loop sweetens the party
	callfunc "HW_Coins", 10;
	specialeffect EF_ANGEL, AREA, strnpcinfo(0);
	hideonnpc strnpcinfo(0);
	mapannounce "hw_town", "[Casper]: A little one is on the way home! Lost Children found: " + $@HW_Kids + "/6. Have some candy, everyone!", bc_map, "0xFF8C00";
	return;
}

// A candy bowl on a doorstep: one knock, a treat... or a trick.
//   callfunc "HW_Bowl", <0..7>;   (from the bowl's own NPC)
function	script	HW_Bowl	{
	.@b = getarg(0);
	if (!$@HW_Active || !callfunc("HW_InRun") || !($@HW_BowlOut & (1 << .@b))) return;
	$@HW_BowlOut = $@HW_BowlOut & ~(1 << .@b);
	getmapxy(.@m$, .@x, .@y, BL_NPC, strnpcinfo(0));
	hideonnpc strnpcinfo(0);
	.@r = rand(100);
	if (.@r < 35) {
		// a treat: candy for the whole party, and coins for the knocker
		$@HW_Bless = 1;
		getitem 6658, 15;
		dispbottom "[Halloween Town] TREAT! A fistful of candy and 15 Halloween Coins!", 0xFF8C00;
	} else if (.@r < 55) {
		// a big treat: coins for everyone, and the knocker is patched up
		callfunc "HW_Coins", 40;
		percentheal 100, 100;
		mapannounce "hw_town", "[Casper]: " + strcharinfo(0) + " found a bowl of real treasure! 40 coins for everyone!", bc_map, "0xFF8C00";
	} else if (.@r < 80) {
		// a trick: something wakes up in the bowl
		.@h = ($@HW_Horde ? $@HW_Horde : 1);
		callsub L_Champion, .@h, rand(3), .@x, .@y;
		dispbottom "[Halloween Town] TRICK! Something was hiding in the candy...", 0xFF4444;
	} else {
		// a trick: a pumpkin bomb
		specialeffect2 EF_SUI_EXPLOSION;
		percentheal -40, 0;
		sc_start SC_STUN, 2000, 1;
		dispbottom "[Halloween Town] TRICK! The pumpkin explodes in your face!", 0xFF4444;
	}
	return;

L_Champion:
	callfunc "HW_Champion", getarg(0), getarg(1), getarg(2), getarg(3);
	return;
}

// Spawns a Jack-o'-Champion of horde <1|2|3>, slot <0..2>, on <x>,<y>:
// crowned, large, 6x the HP. During a horde it counts for 10 of its kind.
function	script	HW_Champion	{
	switch (($@HW_Tier - 1) * 3 + getarg(0)) {
	case 1: setarray .@id[0], 1163, 1297, 1208; break;
	case 2: setarray .@id[0], 1268, 1219, 1302; break;
	case 3: setarray .@id[0], 1283, 1120, 1720; break;
	case 4: setarray .@id[0], 3200, 2993, 2959; break;
	case 5: setarray .@id[0], 20360, 2736, 3480; break;
	case 6: setarray .@id[0], 3193, 2474, 20367; break;
	case 7: setarray .@id[0], 20420, 3474, 3041; break;
	case 8: setarray .@id[0], 3448, 3039, 20616; break;
	case 9: setarray .@id[0], 20920, 3091, 20939; break;
	default: return;
	}
	.@mob = .@id[getarg(1)];
	.@x = getarg(2);
	.@y = getarg(3);
	areamonster "hw_town", .@x - 2, .@y - 2, .@x + 2, .@y + 2, "Jack-o'-Champion " + getmonsterinfo(.@mob, MOB_NAME), .@mob, 1, "HW_Engine::OnChampKill" + getarg(1), Size_Large;
	.@g = $@mobid[0];
	if (!.@g || !unitexists(.@g)) return;
	.@hp = getmonsterinfo(.@mob, MOB_MAXHP) * 6;
	setunitdata .@g, UMOB_MAXHP, .@hp;
	setunitdata .@g, UMOB_HP, .@hp;
	hateffect HAT_EF_SUBJECT_AURA_GOLD, true, .@g;
	return;
}

// The run's members who are on hw_town right now (the solo player, or every
// party member). Fills $@HW_MA[] (account ids) / $@HW_MC[] (char ids) and
// returns how many.
function	script	HW_Members	{
	deletearray $@HW_MA[0], 128;
	deletearray $@HW_MC[0], 128;
	.@n = 0;
	if ($@HW_PID) {
		getpartymember $@HW_PID, 1;
		copyarray .@cid[0], $@partymembercid[0], $@partymembercount;
		getpartymember $@HW_PID, 2;
		copyarray .@aid[0], $@partymemberaid[0], $@partymembercount;
		.@total = $@partymembercount;
		for (.@i = 0; .@i < .@total; .@i++) {
			if (!isloggedin(.@aid[.@i], .@cid[.@i])) continue;
			if (strcharinfo(3, .@cid[.@i]) != "hw_town") continue;
			$@HW_MA[.@n] = .@aid[.@i];
			$@HW_MC[.@n] = .@cid[.@i];
			.@n++;
		}
	} else if (isloggedin($@HW_AID, $@HW_CID) && strcharinfo(3, $@HW_CID) == "hw_town") {
		$@HW_MA[0] = $@HW_AID;
		$@HW_MC[0] = $@HW_CID;
		.@n = 1;
	}
	return .@n;
}

// Gives <n> Halloween Coins to every member on hw_town, whoever made the kill
function	script	HW_Coins	{
	.@n = callfunc("HW_Members");
	for (.@i = 0; .@i < .@n; .@i++)
		getitem 6658, getarg(0), $@HW_MA[.@i];
	return;
}

// Only the run's owner (party leader / solo player) drives the story.
// Returns 1 for them; anyone else is told so (party members) or ignored.
function	script	HW_IsOwner	{
	if (getcharid(0) == $@HW_CID) return 1;
	if ($@HW_PID && getcharid(1) == $@HW_PID) {
		mes "[" + getarg(0) + "]";
		mes "Only your party leader, ^0000FF" + $@HW_Name$ + "^000000, can speak for the party tonight.";
		close2;
	}
	return 0;
}

// MVP id for <1 MVP 1 | 2 MVP 2 | 3 final> on the current moon
function	script	HW_Boss	{
	setarray .@b[0], 1039, 2483, 20520,	// MVP 1: Baphomet, Nightmare Baphomet, Chaos Baphomet
	                 1492, 1832, 20928,	// MVP 2: Samurai Specter, Ifrit, The One
	                 1418, 3029, 20943;	// Final: Evil Snake Lord, Reaper Yanku, Death Witch
	return .@b[(getarg(0) - 1) * 3 + $@HW_Tier - 1];
}

// Spawns MVP <1|2> around the run's owner

}

// Blair's hoard for the attached character, on the current moon
function	script	HW_GiveRewards	{
	setarray .@coin[1], 100, 200, 300;
	setarray .@egg[1], 5, 10, 15;
	setarray .@box[1], 1, 3, 5;
	setarray .@zeny[1], 1500000, 3000000, 5000000;
	setarray .@bexp[1], 50000000, 2000000000, 30000000000;
	setarray .@jexp[1], 30000000, 1200000000, 18000000000;
	.@t = $@HW_Tier;
	// items missing on this server (e.g. on Pre-renewal) are skipped
	if (getitemname(6658) != "null") getitem 6658, .@coin[.@t];
	if (getitemname(12990) != "null") getitem 12990, .@egg[.@t];
	if (getitemname(22669) != "null") getitem 22669, .@box[.@t];
	Zeny += .@zeny[.@t];
	getexp .@bexp[.@t], .@jexp[.@t];
	// the grade: S a Halloween Box and 100 coins, A 50 coins, B 20 coins
	.@gc = 0;
	if ($@HW_Grade$ == "S") {
		.@gc = 100;
		if (getitemname(22669) != "null") getitem 22669, 1;
	} else if ($@HW_Grade$ == "A") .@gc = 50;
	else if ($@HW_Grade$ == "B") .@gc = 20;
	// all six Lost Children home: a Halloween Box and 50 coins more
	if ($@HW_Kids >= 6) {
		.@gc += 50;
		if (getitemname(22669) != "null") getitem 22669, 1;
	}
	if (.@gc && getitemname(6658) != "null") getitem 6658, .@gc;
	// a personal best
	.@time = gettimetick(2) - $@HW_RunID;
	if (!HW_Best || .@time < HW_Best) HW_Best = .@time;
	return;
}

	for (.@l = 0; .@l < 4; .@l++) hideonnpc "Jack-o'-Lantern#HWL" + .@l;
	$@HW_FinalGID = 0;
	$@HW_ShieldBack = 0;
	$@HW_Stage = 13;
	movenpc "Blair#HW", 261, 158;
	hideoffnpc "Blair#HW";
	mapannounce "hw_town", "[Casper]: She's... she's not a monster anymore. Look at the portal!", bc_map, "0xFF8C00";
	$@HW_Guide = 4;
	end;

// The rest of the party's share of Blair's hoard (the owner got theirs in
// the box's own script)
OnRewardParty:
	sleep 1000;	// let the box's script run on before attaching to anyone
	.@n = callfunc("HW_Members");
	copyarray .@aid[0], $@HW_MA[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@aid[.@i] == $@HW_AID) continue;
		if (!attachrid(.@aid[.@i])) continue;
		callfunc "HW_GiveRewards";
		dispbottom "[Halloween Town] You received Blair's hoard. Happy Halloween!";
		detachrid;
	}
	end;

// Casper flies everyone home: between him and the Halloween Exchanger
OnFinish:
	sleep 1500;
	mapwarp "hw_town", "niflheim", 182, 190;
	donpcevent "HW_Engine::OnCleanup";
	end;

// Casper pops up beside the run's owner
L_CasperToOwner:
	.@x = 261;
	.@y = 158;
	getmapxy(.@m$, .@px, .@py, BL_PC, $@HW_Name$);
	if (.@m$ == "hw_town") { .@x = .@px + 1; .@y = .@py; }
	movenpc "Casper#HW", .@x, .@y;
	hideoffnpc "Casper#HW";
	return;

// Moves every monster within 10 cells of the sky portal (261,158) to a random
// walkable spot elsewhere in town
L_SafeZone:
	.@n = getareaunits(BL_MOB, "hw_town", 251, 148, 271, 168, .@mob);
	for (.@i = 0; .@i < .@n; .@i++) {
		for (.@try = 0; .@try < 20; .@try++) {
			.@x = rand(90, 344);
			.@y = rand(46, 358);
			if (.@x >= 241 && .@x <= 281 && .@y >= 138 && .@y <= 178) continue;	// well away from the portal
			if (checkcell("hw_town", .@x, .@y, cell_chkpass)) break;
		}
		if (.@try < 20) unitwarp .@mob[.@i], "hw_town", .@x, .@y;
	}
	return;

// Blocks (false) or opens (true) the ring of cells under the pumpkins:
// every cell exactly 3 cells (square distance) from the portal, 261,158
L_Barrier:
	for (.@dx = -3; .@dx <= 3; .@dx++)
		for (.@dy = -3; .@dy <= 3; .@dy++)
			if (.@dx == -3 || .@dx == 3 || .@dy == -3 || .@dy == 3)
				setcell "hw_town", 261 + .@dx, 158 + .@dy, 261 + .@dx, 158 + .@dy, cell_walkable, getarg(0);
	return;

// 1-second loop: HUD for every member, Casper's pop-ups, time limit
OnTimer1000:
	if (!$@HW_Active) {
		stopnpctimer;
		end;
	}
	.@rem = $@HW_TimeLimit - gettimetick(2);
	if (.@rem <= 0) {
		stopnpctimer;
		mapannounce "hw_town", "[Casper]: Midnight is over! The portal is closing, hold on to your broom!", bc_map, "0xFF0000";
		sleep 2000;
		mapwarp "hw_town", "niflheim", 182, 190;
		donpcevent "HW_Engine::OnCleanup";
		end;
	}
	// Safe zone: while monsters roam the town (the hordes, and the road to the
	// old hill), none may stay by the sky portal, where players arrive and come
	// back after dying. They are moved somewhere else in town, not killed, so
	// nothing is counted. Off for the final fight, which happens at the portal.
	if ($@HW_Stage == 1 || $@HW_Stage == 5 || $@HW_Stage == 8 || $@HW_Stage == 9)
		callsub L_SafeZone;

	// candy for the party (a child found, or a treat from a bowl)
	if ($@HW_Bless) {
		$@HW_Bless = 0;
		.@n = callfunc("HW_Members");
		copyarray .@aid[0], $@HW_MA[0], .@n;
		for (.@i = 0; .@i < .@n; .@i++) {
			if (!attachrid(.@aid[.@i])) continue;
			if (Hp > 0) {
				sc_start SC_BLESSING, 180000, 10;
				sc_start SC_INCREASEAGI, 180000, 10;
				specialeffect2 EF_BLESSING;
			}
			detachrid;
		}
	}

	// the Lost Children: 2 in each horde (at 30% and 70%); one left alone 4
	// minutes fades into the shadows
	if ($@HW_Stage == 1 || $@HW_Stage == 5 || $@HW_Stage == 9) {
		.@pct = ($@HW_K[0] + $@HW_K[1] + $@HW_K[2]) * 100 / ($@HW_Goal * 3);
		.@c1 = $@HW_Horde * 2 - 1;
		.@c2 = $@HW_Horde * 2;
		if (.@pct >= 30 && !($@HW_KidSeen & (1 << .@c1))) callsub L_ChildAppear, .@c1;
		if (.@pct >= 70 && !($@HW_KidSeen & (1 << .@c2))) callsub L_ChildAppear, .@c2;
	}
	for (.@c = 1; .@c <= 6; .@c++)
		if (($@HW_KidOut & (1 << .@c)) && gettimetick(2) >= $@HW_KidT[.@c] + 240) {
			$@HW_KidOut = $@HW_KidOut & ~(1 << .@c);
			hideonnpc "Lost Child#HWC" + .@c;
			mapannounce "hw_town", "[Casper]: No... the shadows took the little one.", bc_map, "0xFF4444";
		}

	// the final fight: 60 seconds after the 4th lantern, the shadows snuff them
	if ($@HW_Stage == 12 && $@HW_ShieldBack && gettimetick(2) >= $@HW_ShieldBack) {
		$@HW_ShieldBack = 0;
		$@HW_LitMask = 0;
		$@HW_Lit = 0;
		if ($@HW_FinalGID && unitexists($@HW_FinalGID)) setunitdata $@HW_FinalGID, UMOB_DMGIMMUNE, 1;
		for (.@l = 0; .@l < 4; .@l++) setnpcdisplay "Jack-o'-Lantern#HWL" + .@l, "Jack-o'-Lantern", 10174;
		mapannounce "hw_town", "[Casper]: The shadows snuffed out the lanterns! Light them again!", bc_map, "0xFF4444";
	}

	.@time$ = (.@rem / 60) + ":" + ((.@rem % 60) < 10 ? "0" : "") + (.@rem % 60);
	switch ($@HW_Stage) {
	case 0:  .@info$ = "Talk to Casper"; break;
	case 1:
	case 5:
	case 9:  .@info$ = "Horde " + $@HW_Horde + " | " + $@HW_K[0] + "/" + $@HW_Goal + " " + $@HW_K[1] + "/" + $@HW_Goal + " " + $@HW_K[2] + "/" + $@HW_Goal; break;
	case 2:  .@info$ = "Find Blair in the cemetery"; break;
	case 3:
	case 7:
	case 12: .@info$ = "Defeat the MVP"; break;
	case 4:  .@info$ = "Find Casper by the ruined house"; break;
	case 6:
	case 10: .@info$ = "Talk to Casper"; break;
	case 8:  .@info$ = "Find Blair on the old hill"; break;
	case 11: .@info$ = "Stop Blair at the sky portal"; break;
	case 13: .@info$ = "Talk to Blair at the sky portal"; break;
	default: .@info$ = "Open Blair's hoard";
	}
	.@n = callfunc("HW_Members");
	copyarray .@aid[0], $@HW_MA[0], .@n;
	setarray .@guide$[1], "OnBlair1", "OnHouse", "OnBlair2", "OnSoul";
	if ($@HW_Stage >= 1) .@info$ += " | Children " + $@HW_Kids + "/6";
	if ($@HW_Stage == 12) .@info$ = ($@HW_ShieldBack ? "Strike Blair now! (" + ($@HW_ShieldBack - gettimetick(2)) + "s)" : "Light the lanterns " + $@HW_Lit + "/4");
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aid[.@i])) continue;
		if ($@HW_KidPoint) addtimer 500, "HW_Guide::OnChild";
		showscript "[Halloween Town] " + .@time$ + " | " + .@info$;
		if ($@HW_Guide && getcharid(0) == $@HW_CID) {
			addtimer 100, "HW_Guide::" + .@guide$[$@HW_Guide];
			$@HW_Guide = 0;
		}
		if (.@rem == 1800 || .@rem == 600 || .@rem == 300 || .@rem == 60)
			dispbottom "[Halloween Town] " + (.@rem / 60) + " minute(s) left before the portal closes!";
		detachrid;
	}
	$@HW_KidPoint = 0;
	setnpctimer 0;
	end;

// A Lost Child appears somewhere in town, away from the portal; Casper hears
// them (the loop hands his pointer to everyone)
L_ChildAppear:
	.@c = getarg(0);
	$@HW_KidSeen = $@HW_KidSeen | (1 << .@c);
	for (.@try = 0; .@try < 50; .@try++) {
		getfreecell("hw_town", .@x, .@y, 217, 202, 127, 156);
		if (distance(.@x, .@y, 261, 158) >= 25) break;
	}
	if (.@try >= 50) return;
	$@HW_KidOut = $@HW_KidOut | (1 << .@c);
	$@HW_KidT[.@c] = gettimetick(2);
	movenpc "Lost Child#HWC" + .@c, .@x, .@y;
	hideoffnpc "Lost Child#HWC" + .@c;
	$@HW_KidNow = .@c;
	$@HW_KidX = .@x;
	$@HW_KidY = .@y;
	$@HW_KidPoint = 1;
	mapannounce "hw_town", "[Casper]: I hear a child crying somewhere in town! Find them!", bc_map, "0xFF8C00";
	return;

OnCleanup:
	stopnpctimer;
	killmonster "hw_town", "All";
	hideonnpc "Halloween Box#HW";
	movenpc "Casper#HW", 255, 154;
	hideoffnpc "Casper#HW";
	movenpc "Blair#HW", 202, 240;
	hideoffnpc "Blair#HW";
	for (.@i = 0; .@i < 8; .@i++)
		hideoffnpc "Pumpkin#HWB" + .@i;
	callsub L_Barrier, false;
	callsub L_HideExtras;
	$@HW_Active = 0;
	$@HW_CID = 0;
	$@HW_AID = 0;
	$@HW_Name$ = "";
	$@HW_PID = 0;
	$@HW_Tier = 0;
	$@HW_Stage = 0;
	$@HW_Horde = 0;
	$@HW_Goal = 0;
	$@HW_Guide = 0;
	$@HW_RunID = 0;
	$@HW_TimeLimit = 0;
	deletearray $@HW_K[0], 3;
	end;

// .id[0..2], .goal and .spawn for horde <1|2|3> on moon $@HW_Tier.
// Loaded on every use, so nothing depends on OnInit having run.
L_Cfg:
	switch (($@HW_Tier - 1) * 3 + getarg(0)) {
	// Blue Moon
	case 1: setarray .id[0], 1163, 1297, 1208;   .goal = 100; .spawn = 150; break;	// Raydric, Ancient Mummy, Wanderer
	case 2: setarray .id[0], 1268, 1219, 1302;   .goal = 150; .spawn = 200; break;	// Bloody Knight, Abysmal Knight, Dark Illusion
	case 3: setarray .id[0], 1283, 1120, 1720;   .goal = 200; .spawn = 250; break;	// Chimera, Ghostring, Hydrolancer
	// Blood Moon
	case 4: setarray .id[0], 3200, 2993, 2959;   .goal = 150; .spawn = 200; break;	// Wicked Chimera, Malicious Baby Ghost, Torturous Redeemer
	case 5: setarray .id[0], 20360, 2736, 3480;  .goal = 250; .spawn = 300; break;	// Mutant Plaga, Necromancer Ringleader, Immortal Cursed Zombie
	case 6: setarray .id[0], 3193, 2474, 20367;  .goal = 400; .spawn = 450; break;	// Ancient Medium Gigantes, 2nd Commander of Destruction, Contaminated Raydric
	// Eclipse
	case 7: setarray .id[0], 20420, 3474, 3041;  .goal = 250; .spawn = 300; break;	// Corrupted Wanderer, Immortal Cursed Knight, Freezing Morocc Reincarnation
	case 8: setarray .id[0], 3448, 3039, 20616;  .goal = 400; .spawn = 450; break;	// Renovated Amdarais, Blazing Morocc Reincarnation, Bone Ferus
	case 9: setarray .id[0], 20920, 3091, 20939; .goal = 666; .spawn = 700; break;	// Lavaeter, Brinaranea, Lude Gal
	}
	return;

OnInit:
	callsub L_Barrier, false;
	end;
}

// Opens a shop window (item icons, prices in Halloween Coins). The list is
// built when the server starts, from the items this server actually has:
// a prize missing here (e.g. on Pre-renewal) is swapped for a Halloween

niflheim,180,191,4	script	Halloween Exchanger	1130,{
	callfunc "HW_Portrait", 3;
	mes "[Halloween Exchanger]";
	mes "Trick or treat! Got any ^FF8C00Halloween Coins^000000? Blair's ghosts drop them all over Halloween Town.";
	mes "My prices are in coins, not zeny. You have ^0000FF" + countitem(6658) + "^000000 coins.";
	next;
	switch (select("Show me your treats:Trade Halloween Tickets for Cash Points:Never mind")) {
	case 1:
		callfunc "HW_Portrait", 0;
		close2;
		callshop "HW_CoinShop", 1;
		end;
	case 2:
		// 3 Halloween Tickets (7941, from Blair's final form) = 1 Cash Point
		.@have = countitem(7941);
		.@max = .@have / 3;
		mes "[Halloween Exchanger]";
		mes "Ooh, ^FF8C00Halloween Tickets^000000! I'll give you ^0000FF1 Cash Point^000000 for every ^0000FF3 tickets^000000.";
		mes "You have ^0000FF" + .@have + "^000000 tickets, enough for ^0000FF" + .@max + "^000000 Cash Points.";
		if (.@max < 1) {
			mes "Come back with at least 3! Blair drops them when she falls.";
			callfunc "HW_Portrait", 0;
			close;
		}
		next;
		mes "[Halloween Exchanger]";
		mes "How many Cash Points do you want? (1-" + .@max + ")";
		next;
		input .@cp, 1, .@max;
		if (.@cp < 1 || .@cp > .@max || countitem(7941) < .@cp * 3) {
			mes "[Halloween Exchanger]";
			mes "Hehe, nice try. Count your tickets again!";
			callfunc "HW_Portrait", 0;
			close;
		}
		delitem 7941, .@cp * 3;
		#CASHPOINTS += .@cp;
		mes "[Halloween Exchanger]";
		mes "Deal! ^0000FF" + (.@cp * 3) + "^000000 tickets for ^0000FF" + .@cp + "^000000 Cash Points. You have ^0000FF" + #CASHPOINTS + "^000000 Cash Points now. Happy Halloween!";
		callfunc "HW_Portrait", 0;
		close;
	}
	callfunc "HW_Portrait", 0;
	close;

	//                 prize    stand-in  coins
	setarray .@prize[0], 12990,     0,    50,	// Halloween Spooky Pack (Pet_Egg_Scroll7)
	                     22669,     0,   100,	// HALLOWEEN_G_BOX
	                     17317,     0,    50,	// October Spooky Crate (Sweet_Midgard_Scroll)
	                     17307,     0,    50,	// October Spooky Box (Midgard_Scroll)
	                     19910,     0,  1500,	// C_Halloween_Hat
	                    420597, 20172,  1500,	// C_Halloween_Pumpkin -> Costume Pumpkin Hat
	                     19903,     0,  1500,	// C_Witchs_Hat
	                     20588, 20395,  1500,	// C_Nifl_Bloom -> Costume Black Witch Hat
	                     31833, 20098,  1500,	// C_Halloween_Cat_Short -> Costume Vampire Hairband
	                     20543, 20034,  1500,	// C_Halloween_Poring_Bag -> Costume Jack Castle Bat
	                    400431, 20225,  1500,	// C_Sakura_Witch_Hat -> Costume Pumpkin Head
	                     20397, 19917,  1500,	// C_Jakk -> Costume Dark Pumpkin Hat
	                     23719,   616, 150000;	// Sealed_Card2 -> Old Card Album

	npcshopitem "HW_CoinShop", .@list[0], .@cost[0];
	for (.@i = 1; .@i < .@n; .@i++)
		npcshopadditem "HW_CoinShop", .@list[.@i], .@cost[.@i];
	end;
