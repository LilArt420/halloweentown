// Portrait pop-up: 1 = Casper (bottom-right, like Nightwing),
// 2 = Blair (bottom-left, like Leticia), 3 = Jakk the Halloween Exchanger
// (bottom-left), 0 = hide.
// Files: data/texture/ui/illust/hw_casper.bmp, hw_blair.bmp, hw_jakk.bmp
function	script	HW_Portrait	{
	switch (getarg(0)) {
	case 1:  cutin "hw_casper.bmp", 2; break;
	case 2:  cutin "hw_blair.bmp", 0; break;
	case 3:  cutin "hw_jakk.bmp", 0; break;
	default: cutin "", 255;
	}
	return;
}

// A navigation link to a spot on hw_town: callfunc("HW_Navi", "<text>", x, y).
// Clicking it opens the navigation; the coordinates are never shown.
function	script	HW_Navi	{
	return "<NAVI>" + getarg(0) + "<INFO>hw_town," + getarg(1) + "," + getarg(2) + ",0,000,0</INFO></NAVI>";
}

// The run's members who are on hw_town right now (the solo player, or every
// party member). Fills $@HW_MA[] (account ids) / $@HW_MC[] (char ids) and
// returns how many.
function	script	HW_Members	{
	deletearray $@HW_MA[0], 128;
	deletearray $@HW_MC[0], 128;
	.@n = 0;
	if ($@HW_PID) {
		getpartymember $@HW_PID, 1;
		copyarray .@cid[0], $@partymembercid[0], $@partymembercount;
		getpartymember $@HW_PID, 2;
		copyarray .@aid[0], $@partymemberaid[0], $@partymembercount;
		.@total = $@partymembercount;
		for (.@i = 0; .@i < .@total; .@i++) {
			if (!isloggedin(.@aid[.@i], .@cid[.@i])) continue;
			if (strcharinfo(3, .@cid[.@i]) != "hw_town") continue;
			$@HW_MA[.@n] = .@aid[.@i];
			$@HW_MC[.@n] = .@cid[.@i];
			.@n++;
		}
	} else if (isloggedin($@HW_AID, $@HW_CID) && strcharinfo(3, $@HW_CID) == "hw_town") {
		$@HW_MA[0] = $@HW_AID;
		$@HW_MC[0] = $@HW_CID;
		.@n = 1;
	}
	return .@n;
}

// Gives <n> Halloween Coins to every member on hw_town, whoever made the kill
function	script	HW_Coins	{
	.@n = callfunc("HW_Members");
	for (.@i = 0; .@i < .@n; .@i++)
		getitem 6658, getarg(0), $@HW_MA[.@i];
	return;
}

// Only the run's owner (party leader / solo player) drives the story.
// Returns 1 for them; anyone else is told so (party members) or ignored.
function	script	HW_IsOwner	{
	if (getcharid(0) == $@HW_CID) return 1;
	if ($@HW_PID && getcharid(1) == $@HW_PID) {
		mes "[" + getarg(0) + "]";
		mes "Only your party leader, ^0000FF" + $@HW_Name$ + "^000000, can speak for the party tonight.";
		close2;
	}
	return 0;
}

// MVP id for <1 MVP 1 | 2 MVP 2 | 3 final> on the current moon
function	script	HW_Boss	{
	setarray .@b[0], 1039, 2483, 20520,	// MVP 1: Baphomet, Nightmare Baphomet, Chaos Baphomet
	                 1492, 1832, 20928,	// MVP 2: Samurai Specter, Ifrit, The One
	                 1418, 3029, 20943;	// Final: Evil Snake Lord, Reaper Yanku, Death Witch
	return .@b[(getarg(0) - 1) * 3 + $@HW_Tier - 1];
}

// Spawns MVP <1|2> around the run's owner
function	script	HW_BossNearOwner	{
	.@id = callfunc("HW_Boss", getarg(0));
	.@x = 261;
	.@y = 158;
	getmapxy(.@m$, .@px, .@py, BL_PC, $@HW_Name$);
	if (.@m$ == "hw_town") { .@x = .@px; .@y = .@py; }
	areamonster "hw_town", .@x - 3, .@y - 3, .@x + 3, .@y + 3, "--ja--", .@id, 1, "HW_Engine::OnMvp" + getarg(0) + "Kill";
	mapannounce "hw_town", "[Blair]: " + (getarg(0) == 1 ? "Let's see you survive THIS!" : "My servant will drag you into the shadows!"), bc_map, "0xFF0000";
	return;
}

}

// --- Engine: hordes, kill counts, coins, HUD, time limit, cleanup ----------
-	script	HW_Engine	-1,{

// Fresh run: barrier up, NPCs back in place
OnSetup:
	killmonster "hw_town", "All";
	stopnpctimer;
	$@HW_Stage = 0;
	$@HW_Horde = 0;
	$@HW_Guide = 0;
	deletearray $@HW_K[0], 3;
	movenpc "Casper#HW", 255, 154;
	hideoffnpc "Casper#HW";
	movenpc "Blair#HW", 202, 240;
	hideoffnpc "Blair#HW";
	hideonnpc "Halloween Box#HW";
	for (.@i = 0; .@i < 8; .@i++)
		hideoffnpc "Pumpkin#HWB" + .@i;
	callsub L_Barrier, false;
	end;

// Casper lets the pumpkins go
OnRelease:
	for (.@i = 0; .@i < 8; .@i++)
		hideonnpc "Pumpkin#HWB" + .@i;
	callsub L_Barrier, true;
	end;

// Spawns horde $@HW_Horde for the current moon: all of it at once, all over
// the map (no refills)
OnStartHorde:
	callsub L_Cfg, $@HW_Horde;
	deletearray $@HW_K[0], 3;
	$@HW_Goal = .goal;
	for (.@i = 0; .@i < 3; .@i++)
		areamonster "hw_town", 90, 46, 344, 358, "--ja--", .id[.@i], .spawn, "HW_Engine::OnHKill" + .@i;
	mapannounce "hw_town", "[Casper]: Horde " + $@HW_Horde + "! Slay " + .goal + " each of " + getmonsterinfo(.id[0], MOB_NAME) + ", " + getmonsterinfo(.id[1], MOB_NAME) + " and " + getmonsterinfo(.id[2], MOB_NAME) + "!", bc_map, "0xFF8C00";
	end;

OnHKill0:
	callsub L_Kill, 0;
	end;
OnHKill1:
	callsub L_Kill, 1;
	end;
OnHKill2:
	callsub L_Kill, 2;
	end;

// Any member's kill counts; every member on the map gets the coin
L_Kill:
	if (!$@HW_Active || ($@HW_Stage != 1 && $@HW_Stage != 5 && $@HW_Stage != 9)) return;
	.@t = getarg(0);
	callfunc "HW_Coins", 1;
	if ($@HW_K[.@t] < $@HW_Goal) $@HW_K[.@t]++;
	callsub L_Cfg, $@HW_Horde;
	if (playerattached())
		dispbottom "[Halloween Town] " + getmonsterinfo(.id[0], MOB_NAME) + ": " + $@HW_K[0] + "/" + $@HW_Goal + " | " + getmonsterinfo(.id[1], MOB_NAME) + ": " + $@HW_K[1] + "/" + $@HW_Goal + " | " + getmonsterinfo(.id[2], MOB_NAME) + ": " + $@HW_K[2] + "/" + $@HW_Goal;
	if ($@HW_K[0] < $@HW_Goal || $@HW_K[1] < $@HW_Goal || $@HW_K[2] < $@HW_Goal) return;

	killmonster "hw_town", "All";
	if ($@HW_Stage == 1) {
		$@HW_Stage = 2;
		mapannounce "hw_town", "[Casper]: The souls are scattering! Blair is still in the cemetery!", bc_map, "0xFF8C00";
		$@HW_Guide = 1;	// the HUD loop hands it to the owner (see OnTimer1000)
	} else {
		$@HW_Stage = ($@HW_Stage == 5) ? 6 : 10;
		callsub L_CasperToOwner;
		mapannounce "hw_town", "[Casper]: Psst! Over here! Talk to me!", bc_map, "0xFF8C00";
	}
	return;

// The dead wandering town on the way to the old hill (after MVP 2): coins
// only, no count
OnFillerKill:
	if (!$@HW_Active) end;
	callfunc "HW_Coins", 1;
	end;

// MVP 1 down: the hero swears to stop her; Casper waits at his house
OnMvp1Kill:
	if (!$@HW_Active || $@HW_Stage != 3) end;
	callfunc "HW_Coins", 15;
	if (isloggedin($@HW_AID, $@HW_CID))
		unittalk $@HW_AID, $@HW_Name$ + " : I need to stop her, no matter what!";
	$@HW_Stage = 4;
	movenpc "Casper#HW", 236, 229;
	hideoffnpc "Casper#HW";
	mapannounce "hw_town", "[Casper]: Over here, by the old house past the cemetery!", bc_map, "0xFF8C00";
	$@HW_Guide = 2;
	end;

// MVP 2 down: Blair can be reached on the old hill, and the dead of Hordes
// 1 and 2 wander the town on the way
OnMvp2Kill:
	if (!$@HW_Active || $@HW_Stage != 7) end;
	callfunc "HW_Coins", 15;
	$@HW_Stage = 8;
	movenpc "Blair#HW", 336, 328;
	hideoffnpc "Blair#HW";
	// a fifth of each horde's total, scattered in five patches along the way
	// from the owner to the old hill (Blue 210, Red 300, Eclipse 450 in all)
	.@sx = 261;
	.@sy = 158;
	getmapxy(.@m$, .@px, .@py, BL_PC, $@HW_Name$);
	if (.@m$ == "hw_town") { .@sx = .@px; .@sy = .@py; }
	for (.@h = 1; .@h <= 2; .@h++) {
		callsub L_Cfg, .@h;
		for (.@i = 0; .@i < 3; .@i++) {
			.@each = .spawn / 25;	// a fifth of the total, over five patches
			if (.@each < 1) .@each = 1;
			for (.@k = 1; .@k <= 5; .@k++) {
				.@cx = .@sx + (336 - .@sx) * .@k / 6;
				.@cy = .@sy + (328 - .@sy) * .@k / 6;
				areamonster "hw_town", .@cx - 20, .@cy - 20, .@cx + 20, .@cy + 20, "--ja--", .id[.@i], .@each, "HW_Engine::OnFillerKill";
			}
		}
	}
	mapannounce "hw_town", "[Casper]: The way is open! Blair is on the old hill, and the dead are wandering all over town!", bc_map, "0xFF8C00";
	$@HW_Guide = 3;
	end;

// Final MVP down: Blair's cleansed soul at the portal
OnFinalKill:
	if (!$@HW_Active || $@HW_Stage != 12) end;
	callfunc "HW_Coins", 15;
	killmonster "hw_town", "All";
	$@HW_Stage = 13;
	movenpc "Blair#HW", 261, 158;
	hideoffnpc "Blair#HW";
	mapannounce "hw_town", "[Casper]: She's... she's not a monster anymore. Look at the portal!", bc_map, "0xFF8C00";
	$@HW_Guide = 4;
	end;

// The rest of the party's share of Blair's hoard (the owner got theirs in
// the box's own script)
OnRewardParty:
	sleep 1000;	// let the box's script run on before attaching to anyone
	.@n = callfunc("HW_Members");
	copyarray .@aid[0], $@HW_MA[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (.@aid[.@i] == $@HW_AID) continue;
		if (!attachrid(.@aid[.@i])) continue;
		callfunc "HW_GiveRewards";
		dispbottom "[Halloween Town] You received Blair's hoard. Happy Halloween!";
		detachrid;
	}
	end;

// Casper flies everyone home: between him and the Halloween Exchanger
OnFinish:
	sleep 1500;
	mapwarp "hw_town", "niflheim", 182, 190;
	donpcevent "HW_Engine::OnCleanup";
	end;

// Casper pops up beside the run's owner
L_CasperToOwner:
	.@x = 261;
	.@y = 158;
	getmapxy(.@m$, .@px, .@py, BL_PC, $@HW_Name$);
	if (.@m$ == "hw_town") { .@x = .@px + 1; .@y = .@py; }
	movenpc "Casper#HW", .@x, .@y;
	hideoffnpc "Casper#HW";
	return;

// Blocks (false) or opens (true) the ring of cells under the pumpkins:
// every cell exactly 3 cells (square distance) from the portal, 261,158
L_Barrier:
	for (.@dx = -3; .@dx <= 3; .@dx++)
		for (.@dy = -3; .@dy <= 3; .@dy++)
			if (.@dx == -3 || .@dx == 3 || .@dy == -3 || .@dy == 3)
				setcell "hw_town", 261 + .@dx, 158 + .@dy, 261 + .@dx, 158 + .@dy, cell_walkable, getarg(0);
	return;

// 1-second loop: HUD for every member, Casper's pop-ups, time limit
OnTimer1000:
	if (!$@HW_Active) {
		stopnpctimer;
		end;
	}
	.@rem = $@HW_TimeLimit - gettimetick(2);
	if (.@rem <= 0) {
		stopnpctimer;
		mapannounce "hw_town", "[Casper]: Midnight is over! The portal is closing, hold on to your broom!", bc_map, "0xFF0000";
		sleep 2000;
		mapwarp "hw_town", "niflheim", 182, 190;
		donpcevent "HW_Engine::OnCleanup";
		end;
	}
	.@time$ = (.@rem / 60) + ":" + ((.@rem % 60) < 10 ? "0" : "") + (.@rem % 60);
	switch ($@HW_Stage) {
	case 0:  .@info$ = "Talk to Casper"; break;
	case 1:
	case 5:
	case 9:  .@info$ = "Horde " + $@HW_Horde + " | " + $@HW_K[0] + "/" + $@HW_Goal + " " + $@HW_K[1] + "/" + $@HW_Goal + " " + $@HW_K[2] + "/" + $@HW_Goal; break;
	case 2:  .@info$ = "Find Blair in the cemetery"; break;
	case 3:
	case 7:
	case 12: .@info$ = "Defeat the MVP"; break;
	case 4:  .@info$ = "Find Casper by the ruined house"; break;
	case 6:
	case 10: .@info$ = "Talk to Casper"; break;
	case 8:  .@info$ = "Find Blair on the old hill"; break;
	case 11: .@info$ = "Stop Blair at the sky portal"; break;
	case 13: .@info$ = "Talk to Blair at the sky portal"; break;
	default: .@info$ = "Open Blair's hoard";
	}
	.@n = callfunc("HW_Members");
	copyarray .@aid[0], $@HW_MA[0], .@n;
	setarray .@guide$[1], "OnBlair1", "OnHouse", "OnBlair2", "OnSoul";
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aid[.@i])) continue;
		showscript "[Halloween Town] " + .@time$ + " | " + .@info$;
		if ($@HW_Guide && getcharid(0) == $@HW_CID) {
			addtimer 100, "HW_Guide::" + .@guide$[$@HW_Guide];
			$@HW_Guide = 0;
		}
		if (.@rem == 1800 || .@rem == 600 || .@rem == 300 || .@rem == 60)
			dispbottom "[Halloween Town] " + (.@rem / 60) + " minute(s) left before the portal closes!";
		detachrid;
	}
	setnpctimer 0;
	end;

OnCleanup:
	stopnpctimer;
	killmonster "hw_town", "All";
	hideonnpc "Halloween Box#HW";
	movenpc "Casper#HW", 255, 154;
	hideoffnpc "Casper#HW";
	movenpc "Blair#HW", 202, 240;
	hideoffnpc "Blair#HW";
	for (.@i = 0; .@i < 8; .@i++)
		hideoffnpc "Pumpkin#HWB" + .@i;
	callsub L_Barrier, false;
	$@HW_Active = 0;
	$@HW_CID = 0;
	$@HW_AID = 0;
	$@HW_Name$ = "";
	$@HW_PID = 0;
	$@HW_Tier = 0;
	$@HW_Stage = 0;
	$@HW_Horde = 0;
	$@HW_Goal = 0;
	$@HW_Guide = 0;
	$@HW_RunID = 0;
	$@HW_TimeLimit = 0;
	deletearray $@HW_K[0], 3;
	end;

// .id[0..2], .goal and .spawn for horde <1|2|3> on moon $@HW_Tier.
// Loaded on every use, so nothing depends on OnInit having run.
L_Cfg:
	switch (($@HW_Tier - 1) * 3 + getarg(0)) {

	// Blood Moon

	case 5: setarray .id[0], 20360, 2736, 3480;  .goal = 250; .spawn = 300; break;	// Mutant Plaga, Necromancer Ringleader, Immortal Cursed Zombie
	case 6: setarray .id[0], 3193, 2474, 20367;  .goal = 400; .spawn = 450; break;	// Ancient Medium Gigantes, 2nd Commander of Destruction, Contaminated Raydric
	// Eclipse
	case 7: setarray .id[0], 20420, 3474, 3041;  .goal = 250; .spawn = 300; break;	// Corrupted Wanderer, Immortal Cursed Knight, Freezing Morocc Reincarnation
	case 8: setarray .id[0], 3448, 3039, 20616;  .goal = 400; .spawn = 450; break;	// Renovated Amdarais, Blazing Morocc Reincarnation, Bone Ferus

// --- Halloween Exchanger (Jakk) ---------------------------------------------
// Opens a shop window (item icons, prices in Halloween Coins). The list is
// built when the server starts, from the items this server actually has:
// a prize missing here (e.g. on Pre-renewal) is swapped for a Halloween
// stand-in, or left out if that is missing too.
-	itemshop	HW_CoinShop	-1,6658,12990:50

niflheim,180,191,4	script	Halloween Exchanger	1130,{
	callfunc "HW_Portrait", 3;
	mes "[Halloween Exchanger]";
	mes "Trick or treat! Got any ^FF8C00Halloween Coins^000000? Blair's ghosts drop them all over Halloween Town.";
	mes "My prices are in coins, not zeny. You have ^0000FF" + countitem(6658) + "^000000 coins.";
	next;
	if (select("Show me your treats:Never mind") == 2) { callfunc "HW_Portrait", 0; close; }
	callfunc "HW_Portrait", 0;
	close2;
	callshop "HW_CoinShop", 1;
	end;

OnInit:
	//                 prize    stand-in  coins
	setarray .@prize[0], 12990,     0,    50,	// Halloween Spooky Pack (Pet_Egg_Scroll7)
	                     22669,     0,   100,	// HALLOWEEN_G_BOX
	                     19910,     0,  1500,	// C_Halloween_Hat
	                    420597, 20172,  1500,	// C_Halloween_Pumpkin -> Costume Pumpkin Hat
	                     19903,     0,  1500,	// C_Witchs_Hat
	                     20588, 20395,  1500,	// C_Nifl_Bloom -> Costume Black Witch Hat
	                     31833, 20098,  1500,	// C_Halloween_Cat_Short -> Costume Vampire Hairband
	                     20543, 20034,  1500,	// C_Halloween_Poring_Bag -> Costume Jack Castle Bat
	                    400431, 20225,  1500,	// C_Sakura_Witch_Hat -> Costume Pumpkin Head
	                     20397, 19917,  1500,	// C_Jakk -> Costume Dark Pumpkin Hat
	                     23719,   616, 150000;	// Sealed_Card2 -> Old Card Album
	.@n = 0;
	for (.@i = 0; .@i < getarraysize(.@prize); .@i += 3) {
		.@id = .@prize[.@i];
		if (getitemname(.@id) == "null") .@id = .@prize[.@i + 1];
		if (!.@id || getitemname(.@id) == "null") continue;
		if (inarray(.@list[0], .@id) != -1) continue;	// stand-in already listed
		.@list[.@n] = .@id;
		.@cost[.@n] = .@prize[.@i + 2];
		.@n++;
	}
	if (!.@n) end;
	npcshopitem "HW_CoinShop", .@list[0], .@cost[0];
	for (.@i = 1; .@i < .@n; .@i++)
		npcshopadditem "HW_CoinShop", .@list[.@i], .@cost[.@i];
	end;
}
